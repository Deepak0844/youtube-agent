# Cartoon scenes: the yt-toon kit

GENERATED by scripts/sync-toon-dts.ts. Cartoon-shorts' own puppets, set pieces and looks (MIT), as React components. A cartoon scene file
imports them with named imports, one module per file below. Times are beat marks (B5, B5e, B5%60, B5+.3), scene ids (hook, hooke, hook+1) or seconds.

## yt-toon/core

`import { Anim, Camera, Chroma, DotsPattern, ENTRANCE_PRESETS, FlashAt, Impact, LinGrad, Ov, RadGrad, Scene, Shake, Show, ToonFilm, clamp, eio, eout, lerp, trf, useToon } from "yt-toon/core";`

```ts
import React from "react";
import { clamp, eio, eout, lerp, ToonFilm, useToon, type ToonClock, type ToonPalette } from "./Stage";
import { type ToonCharacter, type ToonInput } from "./context";
export { ToonFilm, useToon, eout, eio, clamp, lerp };
export type { ToonClock, ToonPalette, ToonCharacter, ToonInput };
/** The SVG transform string cartoon-shorts uses for a moved piece: translate(x,y) rotate(r) scale(s). */
export declare const trf: (x: number, y: number, s?: number, r?: number) => string;
export declare const ENTRANCE_PRESETS: readonly ["slam", "smash", "up", "down", "left", "right", "fade", "out", "pop", "stamp", "growx", "growy", "nudge", "wipe"];
export type EntrancePreset = (typeof ENTRANCE_PRESETS)[number];
export type Transition = "flash" | "zoom" | "whip" | "glitch" | "cut";
/** A comic sound word: a burst star that pops in at `at` with the word on top, until `until`. Position is the centre of the word (stage units). */
export type SoundWord = {
    word: string;
    x: number;
    y: number;
    rot: number;
    at: string | number;
    until: string | number;
    fill?: string;
    size?: number;
};
/**
 * One scene of the film: shown only between its start and its end (the scene id must be in the timeline), entered with `tr`.
 * `children` are drawn in the scene's SVG layer (1080 x 1920 portrait, 1920 x 1080 landscape) and move with <Camera>; `above` is SVG drawn over
 * that layer which the camera does not move; `words` are comic sound words (burst + word, above the camera); `overlays` are HTML text boxes
 * (<Ov>), drawn over everything, after the sound words.
 */
export declare const Scene: React.FC<{
    id: string;
    tr?: Transition;
    children?: React.ReactNode;
    above?: React.ReactNode;
    words?: SoundWord[];
    overlays?: React.ReactNode;
}>;
/** `slam@B3+.2` style entrance list: one preset at a time or several ("slam" in, "out" later). */
export type EntranceStep = {
    preset: EntrancePreset;
    at: string | number;
    dur?: number;
};
/** An HTML text box over the scene (titles, kickers, labels, cards). `css` is the box's position and look, e.g. "left:0;right:0;top:300px;text-align:center"; `during` shows it only in a window of marks ("B2..B4e"). */
export declare const Ov: React.FC<{
    css?: string;
    cls?: string;
    enter?: EntranceStep[];
    during?: string;
    fit?: boolean;
    children?: React.ReactNode;
}>;
/** Wraps SVG drawing in an entrance preset (slam, pop, up, fade...) at a mark. */
export declare const Anim: React.FC<{
    preset: EntrancePreset;
    at: string | number;
    dur?: number;
    also?: EntranceStep[];
    children?: React.ReactNode;
}>;
/** Shows its SVG children only between two marks. */
export declare const Show: React.FC<{
    from: string | number;
    to: string | number;
    children?: React.ReactNode;
}>;
/** Slow camera push over the scene (or a window of it): scale `from` -> `to` around (cx, cy). Do not combine it with a style's own camera. */
export declare const Camera: React.FC<{
    a?: string | number;
    b?: string | number;
    from?: number;
    to?: number;
    cx?: number;
    cy?: number;
}>;
/** The whole frame shakes at a mark. */
export declare const Shake: React.FC<{
    at: string | number;
    dur?: number;
    amp?: number;
}>;
/** A hit frame: inverted for a few frames, then a colour split. Put <FlashAt> 0.1 s later, not at the same time. */
export declare const Impact: React.FC<{
    at: string | number;
    frames?: number;
}>;
export declare const Chroma: React.FC<{
    at: string | number;
    dur?: number;
    px?: number;
}>;
export declare const FlashAt: React.FC<{
    at: string | number;
    opacity?: number;
    dur?: number;
}>;
export type GradStop = [offset: number, color: string, opacity?: number];
export declare const LinGrad: React.FC<{
    id: string;
    stops: GradStop[];
    x1?: number;
    y1?: number;
    x2?: number;
    y2?: number;
}>;
export declare const RadGrad: React.FC<{
    id: string;
    stops: GradStop[];
    cx?: number;
    cy?: number;
    r?: number;
}>;
export declare const DotsPattern: React.FC<{
    id: string;
    color?: string;
    size?: number;
    r?: number;
    op?: number;
}>;
```

## yt-toon/comic

`import { EASE, Follow, P, Puppet, cycle, mix, pose, track } from "yt-toon/comic";`

```ts
import React from "react";
export type PoseName = "stand" | "ready" | "shoot" | "release" | "block" | "leap" | "layup" | "dunk" | "kneel" | "kneelFace" | "kneelRoar" | "roar" | "flex" | "celebrate" | "sit" | "sitSad" | "carry" | "cradle" | "reach" | "point" | "hold" | "write" | "trophy";
export type FaceName = "neutral" | "focus" | "grit" | "shout" | "roar" | "sad" | "cry" | "smile" | "grin" | "wow" | "worry";
/** Joint angles (degrees) and face of one moment: tor (lean), hd (head tilt), nua/nfa near upper/fore-arm, fua/ffa far, nth/nsh near thigh/shin, fth/fsh far, nft/fft feet, turn (0 side .. 1 front), lift, look, blink, tears, open; face; hand/fhand: "fist" | "open". */
export type Pose = Partial<Record<"tor" | "hd" | "nua" | "nfa" | "fua" | "ffa" | "nth" | "nsh" | "fth" | "fsh" | "nft" | "fft" | "turn" | "lift" | "look" | "blink" | "tears" | "sq" | "open", number>> & {
    face?: FaceName;
    hand?: "fist" | "open";
    fhand?: "fist" | "open";
};
/** Ready-made poses by name. */
export declare const P: Record<PoseName, Pose>;
/** Blend two poses: mix(P.stand, P.point, 0.5). */
export declare const mix: (a: Pose, b: Pose, k: number) => Pose;
/** Walk / run / sprint cycle pose for a phase in radians. */
export declare const cycle: (phase: number, amp?: number, kind?: "run" | "walk" | "sprint") => Pose;
/** A value that moves through keys [[seconds, value, ease?], ...] (numbers, arrays or poses). */
export declare const track: <V>(keys: [number, V, string?][]) => (t: number) => V;
export declare const EASE: Record<string, (x: number) => number>;
/** Pose `over` on top of pose `base`: pose(P.stand, { face: "grin" }). */
export declare const pose: (base: Pose, over: Pose) => Pose;
/** Where a puppet is and how it stands at one moment. `dir` -1 faces left; `s` is the scale. */
export type PuppetFrame = {
    pose: Pose;
    x: number;
    y: number;
    dir?: 1 | -1;
    s?: number;
};
export type PuppetOptions = Record<string, string | number | boolean | null>;
/** Joint names a puppet can report: hand, fhand (far hand), head, neck, hip, foot, ffoot, knee, elbow, shoulder, fshoulder. */
export type Joint = "hand" | "fhand" | "head" | "neck" | "hip" | "foot" | "ffoot" | "knee" | "elbow" | "shoulder" | "fshoulder";
/**
 * One character. `who` takes the look from the series cast (ref/cast.json); `options` is a one-off look (or overrides): the puppet
 * options of cartoon-shorts (skin, skinD, hair, hairStyle, top, shirt, shirtD, tie, pants, shoe, beard, cap, glasses, kid, build...).
 * `id` lets a <Follow> hold a prop at one of its joints.
 */
export declare const Puppet: React.FC<{
    id?: string;
    who?: string;
    options?: PuppetOptions;
    act: (t: number) => PuppetFrame;
}>;
/**
 * Holds its children (a phone, a cup, a letter) at a joint of the puppet with this `id`, drawn where you write it (so put it after
 * the puppet, and after anything that should cover the prop's z-order). `place` gives where it goes at time t: [x, y, scale, rotation].
 * Show or hide the children yourself (style display) for the times the prop is not in hand.
 */
export declare const Follow: React.FC<{
    of: string;
    at?: Joint;
    dx?: number;
    dy?: number;
    s?: number;
    r?: number;
    place?: (t: number, joint: [number, number]) => [number, number, number, number];
    children?: React.ReactNode;
}>;
```
